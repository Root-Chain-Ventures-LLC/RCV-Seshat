! Cisco IOS-XE: Seshat anycast, three nodes, IPv4 and IPv6, one AS for all nodes.
! Documentation-range addresses and private ASNs; replace them.
!
! Hashing: the default CEF load sharing hashes on source and destination address. Keep it that way;
! do NOT add "include-ports" (it would spread one exporter over several nodes):
!   ip cef load-sharing algorithm universal
!
ip prefix-list SESHAT-IN4 seq 10 permit 198.51.100.10/32
ip prefix-list DENY-ALL4 seq 10 deny 0.0.0.0/0 le 32
ipv6 prefix-list SESHAT-IN6 seq 10 permit 2001:DB8:100::10/128
ipv6 prefix-list DENY-ALL6 seq 10 deny ::/0 le 128
!
router bgp 65000
 bgp router-id 192.0.2.1
 bgp log-neighbor-changes
 no bgp default ipv4-unicast
 neighbor SESHAT4 peer-group
 neighbor SESHAT4 remote-as 65010
 neighbor SESHAT4 description Seshat nodes IPv4
 neighbor SESHAT4 timers 3 9
 ! neighbor SESHAT4 password CHANGE-ME
 ! Optional BFD (needs a responder on the node host):
 ! neighbor SESHAT4 fall-over bfd
 neighbor 192.0.2.11 peer-group SESHAT4
 neighbor 192.0.2.12 peer-group SESHAT4
 neighbor 192.0.2.13 peer-group SESHAT4
 neighbor SESHAT6 peer-group
 neighbor SESHAT6 remote-as 65010
 neighbor SESHAT6 description Seshat nodes IPv6
 neighbor SESHAT6 timers 3 9
 ! neighbor SESHAT6 password CHANGE-ME
 neighbor 2001:DB8:FFFF::11 peer-group SESHAT6
 neighbor 2001:DB8:FFFF::12 peer-group SESHAT6
 neighbor 2001:DB8:FFFF::13 peer-group SESHAT6
 !
 address-family ipv4
  neighbor SESHAT4 activate
  neighbor SESHAT4 prefix-list SESHAT-IN4 in
  neighbor SESHAT4 prefix-list DENY-ALL4 out
  maximum-paths 8
 exit-address-family
 !
 address-family ipv6
  neighbor SESHAT6 activate
  neighbor SESHAT6 prefix-list SESHAT-IN6 in
  neighbor SESHAT6 prefix-list DENY-ALL6 out
  maximum-paths 8
 exit-address-family
!

!         show bgp ipv4 unicast summary
